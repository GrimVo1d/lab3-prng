47 constexpr int N_SAMPLES = 20;
48 constexpr std::size_t SIZE = 5000; // не менее 5000 чисел в выборке
49 const int K = pt::stats::sturges_k(SIZE);
50 const double crit = pt::stats::chi_critical_005(K - 1);
51
52 auto run = [&](const std::string& name, auto factory) {
53 for (int i = 0; i < N_SAMPLES; ++i) {
54 auto g = factory(static_cast<std::uint64_t>(i) * 0x9E3779B97F4A7C15ULL + 1);
55 auto data = sample_doubles(g, SIZE);
56 auto s = pt::stats::summarize(data);
57 auto [chi2, df] = pt::stats::chi_square_uniform(data, K);
58 bool ok = chi2 <= crit;
59 out << name << ',' << i << ','
60 << s.mean << ',' << s.std_dev << ',' << s.cv << ','
61 << chi2 << ',' << df << ',' << crit << ','
62 << (ok ? 1 : 0) << '\n';
63 std::cerr << name << "[" << i << "] mean=" << s.mean
64 << " std=" << s.std_dev
65 << " CV=" << s.cv
66 << " chi2=" << chi2 << "/" << crit
67 << (ok ? " OK" : " FAIL") << '\n';
68 }
69 };
70
71 run("lcg", [](std::uint64_t s){ return pt::prng::LCG(s); });
72 run("lfg", [](std::uint64_t s){ return pt::prng::LFG(s); });
73 run("midsq", [](std::uint64_t s){ return pt::prng::MidSquareXor(s); });
74 return 0;
75}
