19inline Summary summarize(const std::vector<double>& x) {
20 Summary s{};
21 if (x.empty()) return s;
22 double sum = 0.0;
23 for (double v : x) sum += v;
24 s.mean = sum / x.size();
25 double q = 0.0;
26 for (double v : x) q += (v - s.mean) * (v - s.mean);
27 s.variance = q / x.size();
28 s.std_dev = std::sqrt(s.variance);
29 s.cv = (s.mean != 0.0) ? (s.std_dev / s.mean) : 0.0;
30 return s;
31}

37inline std::pair<double, int> chi_square_uniform(
38 const std::vector<double>& x, int k) {
39 std::vector<long long> n_i(k, 0);
40 for (double v : x) {
41 int idx = std::min(static_cast<int>(v * k), k - 1);
42 if (idx < 0) idx = 0;
43 ++n_i[idx];
44 }
45 double expected = static_cast<double>(x.size()) / k;
46 double chi2 = 0.0;
47 for (int i = 0; i < k; ++i) {
48 double d = n_i[i] - expected;
49 chi2 += d * d / expected;
50 }
51 return {chi2, k - 1};
52}
